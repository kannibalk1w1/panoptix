cmake_minimum_required(VERSION 3.22)
project(panoptixclick VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(BUILD_SHARED_LIBS ON)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})
set(CMAKE_AUTOMOC ON)
set(QT_DEFAULT_MAJOR_VERSION 6)
find_package(ECM REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
include(KDEInstallDirs6)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Quick Network Widgets)
find_package(KWin REQUIRED)
kcoreaddons_add_plugin(panoptixclick SOURCES panoptixclick.cpp INSTALL_NAMESPACE "kwin/effects/plugins")
set_target_properties(panoptixclick PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
target_link_libraries(panoptixclick PRIVATE KWin::kwin Qt6::Core Qt6::Network)
